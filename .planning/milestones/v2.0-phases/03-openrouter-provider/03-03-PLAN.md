def _openrouter_resp_ok(artifact: dict, model: str = "google/gemini-3.1-pro-preview", cost: float = 0.001):
    """Build a /chat/completions response with finish_reason='stop' (STRING, not enum)."""
    from unittest.mock import MagicMock
    import json as _json
    resp = MagicMock()
    choice = MagicMock()
    choice.finish_reason = "stop"                          # STRING — differs from Gemini enum
    choice.message.content = _json.dumps(artifact)
    resp.choices = [choice]
    resp.model = model
    usage = MagicMock()
    usage.cost = cost                                      # BODY field (Pitfall 5)
    resp.usage = usage
    return resp
