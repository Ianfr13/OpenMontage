```python
def synthesize_pipeline(
    analysis: dict[str, Any],
    mode: Literal["template", "replica"] = "template",
    use_llm_fill: bool | None = None,
) -> dict[str, Any]:
    """Match a base pipeline, write staging YAML, return synthesis run record.
    Returns dict with keys: base_pipeline, match_score, mode, staging_path,
    diff_against_base, validation_status, source_analysis_checksum, provider_used,
    slug."""

def accept_synthesis(slug: str) -> tuple[Path, dict[str, Any]]:
    """Promote pipeline_defs/_staging/<slug>.yaml -> pipeline_defs/<slug>.yaml.
    Returns (final_path, updated_record). The ONLY authorized write into
    pipeline_defs/."""

def reject_synthesis(slug: str) -> dict[str, Any]:
    """Delete pipeline_defs/_staging/<slug>.yaml. Emits a rejection record
    with validation_status='invalid' (schema has no 'rejected')."""
```
