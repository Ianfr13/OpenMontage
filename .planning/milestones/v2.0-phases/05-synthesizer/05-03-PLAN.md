       ```python
       def fill_stage_details(base_manifest: dict, analysis: dict, *, mode: str = "template") -> dict:
           """LLM-assisted stage detail fill.

           Returns a manifest with `tools_available` / `review_focus` / `success_criteria`
           populated on stages that already exist in base_manifest. Never invents stages.
           Falls back to base_manifest unchanged on any failure (key missing, env opt-out,
           openai raise, JSON parse failure x2, post-merge validation failure).

           Args:
               base_manifest: Loaded base pipeline manifest (from load_pipeline()).
               analysis: The video_analysis artifact.
               mode: "template" (generalizable) or "replica" (closer reproduction).

           Returns:
               A new manifest dict (deep-copied; original untouched).
           """
           if os.environ.get("VIDEO_SYNTH_LLM_FILL", "true").lower() == "false":
               logger.info("llm_fill: disabled by VIDEO_SYNTH_LLM_FILL=false")
               return deepcopy(base_manifest)
           if not os.environ.get("OPENROUTER_API_KEY"):
               logger.info("llm_fill: OPENROUTER_API_KEY missing — returning base manifest")
               return deepcopy(base_manifest)

           model = os.environ.get("VIDEO_SYNTH_LLM_MODEL", MODEL_DEFAULT)
           prompt = _build_prompt(base_manifest, analysis, mode=mode)

           client = OpenAI(
               base_url="https://openrouter.ai/api/v1",
               api_key=os.environ["OPENROUTER_API_KEY"],
           )

           filled = _call_with_retry(client, model, prompt)
           if filled is None:
               return deepcopy(base_manifest)

           merged = _merge_stage_details(base_manifest, filled)

           # Post-merge semantic validation — if LLM injected a bogus tool, revert.
           from lib.pipeline_synthesizer import validate_synthesized_pipeline
           issues = validate_synthesized_pipeline(merged)
           if issues:
               logger.warning("llm_fill: post-merge validation failed (%d issues) — reverting", len(issues))
               return deepcopy(base_manifest)
           return merged
       ```

    2. `_call_with_retry(client, model, prompt) -> dict | None`:
       Two attempts. Returns parsed JSON dict on success; None on both-failed.

       ```python
       def _call_with_retry(client: OpenAI, model: str, prompt: str) -> dict | None:
           for attempt in (1, 2):
               try:
                   resp = client.chat.completions.create(
                       model=model,
                       max_tokens=MAX_TOKENS,
                       response_format={"type": "json_object"},
                       messages=[{"role": "user", "content": prompt}],
                   )
                   text = (resp.choices[0].message.content or "").strip()
                   return json.loads(text)
               except (APIError, Exception) as exc:
                   logger.warning("llm_fill: attempt %d failed (%s: %s)",
                                  attempt, type(exc).__name__, exc)
                   if attempt == 2:
                       return None
                   continue
           return None
       ```

    3. `_build_prompt(base_manifest, analysis, *, mode) -> str`:
       Composes a prompt that:
       - Names the base pipeline
       - Shows each stage's `name` + existing fields as context
       - Asks for a JSON object with shape:
         ```json
         {"stages": [{"name": "<stage-name>",
                      "tools_available": [...],
                      "review_focus": [...],
                      "success_criteria": [...]}, ...]}
         ```
       - Branches on mode — template adds "Keep values generalizable — this is a reusable template."; replica adds "Aim for closer reproduction of the reference video's specific style."
       - Includes pacing + narrative signals from `analysis` as hints.

    4. `_merge_stage_details(base_manifest, filled) -> dict`:
       - `deepcopy(base_manifest)`; iterate `merged["stages"]`.

       - **Never** touch non-fillable keys (name, skill, agent, required_artifacts_in, produces, etc.).
       - Return the merged manifest.

    ```python
    import shutil

    def accept_synthesis(slug: str) -> tuple[Path, dict[str, Any]]:
        """Promote a staged pipeline to pipeline_defs/.

        Moves pipeline_defs/_staging/<slug>.yaml → pipeline_defs/<slug>.yaml.
        Re-runs semantic validation on the promoted file and emits a schema-validated
        pipeline_synthesis run record.

        Returns:
            (promoted_path, record)

        Raises:
            FileNotFoundError: staging file missing.
            FileExistsError: target under pipeline_defs/ already exists.
        """
        src = STAGING_DIR / f"{slug}.yaml"
        dst = PIPELINE_DEFS_DIR / f"{slug}.yaml"
        if not src.exists():
            raise FileNotFoundError(f"Staging file not found: {src}")
        if dst.exists():
            raise FileExistsError(f"Target already exists (will not overwrite): {dst}")
        # Sibling dirs under the same parent — os.rename works; shutil.move handles fallback.
        shutil.move(str(src), str(dst))

        # Re-validate the promoted file.
        promoted_manifest = load_pipeline(slug)
        issues = validate_synthesized_pipeline(promoted_manifest)
        status = "valid" if not issues else "invalid"

        record = {
            "version": "1.0",
            "base_pipeline": slug,  # caller may rewrite if they retain match data
            "match_score": 0.0,     # accept API is flow-agnostic; meta skill can overwrite
            "mode": "template",
            "staging_path": str(src.relative_to(PIPELINE_DEFS_DIR.parent)),
            "diff_against_base": "",
            "validation_status": status,
            "source_analysis_checksum": "post-accept",
            "provider_used": "gemini",
            "created_at": datetime.now(timezone.utc).isoformat(timespec="seconds"),
        }
        import jsonschema
        jsonschema.validate(instance=record, schema=_load_synthesis_schema())
        return dst, record


    def reject_synthesis(slug: str) -> dict[str, Any]:
        """Delete pipeline_defs/_staging/<slug>.yaml and emit a rejection record.

        Returns the run record (validation_status='invalid' per Pitfall 2 — user
        rejection encoded as the 'invalid' sentinel; schema enum does not include
        'rejected'). Does NOT write the record to disk — caller handles persistence.
        """
        src = STAGING_DIR / f"{slug}.yaml"
        if not src.exists():
            raise FileNotFoundError(f"Staging file not found: {src}")
        src.unlink()
        record = {
            "version": "1.0",
            "base_pipeline": slug,
            "match_score": 0.0,
            "mode": "template",
            "staging_path": str(src.relative_to(PIPELINE_DEFS_DIR.parent)),
            "diff_against_base": "",
            "validation_status": "invalid",   # user rejection — Pitfall 2
            "source_analysis_checksum": "post-reject",
            "provider_used": "gemini",
            "created_at": datetime.now(timezone.utc).isoformat(timespec="seconds"),
        }
        import jsonschema
        jsonschema.validate(instance=record, schema=_load_synthesis_schema())
        return record
    ```
