def _relative_div(a: float, b: float) -> float:
    """Relative divergence. abs(a-b) / max(a,b). Returns 0.0 if both zero."""
    m = max(abs(a), abs(b))
    if m == 0.0:
        return 0.0
    return abs(a - b) / m


def _within_tolerance(a: float, b: float, tol: float = CPM_TOLERANCE) -> bool:
    return _relative_div(a, b) <= tol
